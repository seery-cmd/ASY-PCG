
1.解压：tar -zxvf petsc-3.21.5.tar.gz
2.配置环境变量
export PETSC_DIR=/home/xiaogq/software/petsc-3.21.5
export PETSC_ARCH=test
3.配置,这里自动下载MPI刀test文件
./configure --with-cc=gcc --with-cxx=g++ --with-fc=gfortran --download-mpich --download-fblaslapack

./configure --with-cc=mpicc --with-cxx=mpicxx --with-fc=mpif90 --with-mpi-dir=/home/xiaogq/oyh/software/mpich-4.2.2

./configure --with-precision=__float128 --download-f2cblaslapack  #换精度,加上之前的一些操作,lapack和blas两个库

petsc默认为double数据类型，需要重写编译为float
文档中说明了--with-scalar-type=real/complex   实数/复数
--with-precision=single,double,__float128,__fp16   单精度32/双精度64/128位宽/16位宽

4.编译
make PETSC_DIR=/path/to/download/petsc-3.21.5 PETSC_ARCH=test all

5.检查
make PETSC_DIR=/path/to/download/petsc-3.21.5 PETSC_ARCH=test check

6.bashrc修改,makefile -I导入两个文件
export PETSC_DIR=/home/xiaogq/software/petsc-3.21.5
export PETSC_LIB=/home/xiaogq/software/petsc-3.21.5/lib
